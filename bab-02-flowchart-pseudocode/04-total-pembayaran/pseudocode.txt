ALGORITMA Total_Pembayaran

DEKLARASI
    harga_satuan  : real
    jumlah_barang : integer
    biaya_layanan : real
    subtotal      : real
    total         : real

DESKRIPSI
    read(harga_satuan)
    read(jumlah_barang)
    read(biaya_layanan)

    subtotal <- harga_satuan * jumlah_barang
    total    <- subtotal + biaya_layanan

    write(subtotal)
    write(total)
